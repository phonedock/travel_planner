☁️ 내 구글 드라이브 DB 서버 설정 방법

  1. 본인 구글 계정으로 script.google.com 접속 후 [새 프로젝트]를 만듭니다.
  2. 기존 코드를 모두 지우고 아래 스크립트를 붙여넣은 뒤 저장합니다.
  3. 우측 상단 [배포] → [새 배포] → 유형: 웹 앱 선택
  4. 다음 사용자 권한으로 실행: 나(본인 이메일) / 액세스 권한: 모든 사용자로 설정 후 배포합니다.
  5. 생성된 웹 앱 URL을 복사해 설정창(또는 코드의 DEFAULT_GDRIVE_WEBAPP_URL)에 넣으면 모든 방문자가 내 드라이브의 '여행 사이트 백업' 폴더를 사용하게 됩니다.
const FOLDER_NAME = '여행 사이트 백업';
const FILE_PREFIX = 'travel_planner_backup_';
const MAX_BACKUPS = 5;

function getOrCreateFolder() {
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  return folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER_NAME);
}

function getSortedBackupFiles(folder) {
  const files = folder.getFiles();
  const list = [];
  while (files.hasNext()) {
    const f = files.next();
    if (!f.isTrashed() && f.getName().endsWith('.json')) {
      list.push(f);
    }
  }
  list.sort((a, b) => b.getLastUpdated().getTime() - a.getLastUpdated().getTime());
  return list;
}

function doGet(e) {
  const folder = getOrCreateFolder();
  const list = getSortedBackupFiles(folder);
  const cb = e && e.parameter && e.parameter.callback;
  if (list.length === 0) {
    const errOut = JSON.stringify({ ok: false, error: 'NO_BACKUP' });
    return cb
      ? ContentService.createTextOutput(cb + '(' + errOut + ')').setMimeType(ContentService.MimeType.JAVASCRIPT)
      : ContentService.createTextOutput(errOut).setMimeType(ContentService.MimeType.JSON);
  }
  const latest = list[0];
  const content = latest.getBlob().getDataAsString('UTF-8') || '{}';
  const okOut = JSON.stringify({ ok: true, fileName: latest.getName(), data: JSON.parse(content) });
  return cb
    ? ContentService.createTextOutput(cb + '(' + okOut + ')').setMimeType(ContentService.MimeType.JAVASCRIPT)
    : ContentService.createTextOutput(okOut).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const folder = getOrCreateFolder();
  const raw = e.postData ? e.postData.contents : '{}';
  const parsed = JSON.parse(raw);
  const payload = parsed.data ? parsed.data : parsed;
  const ts = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyyMMdd_HHmmss');
  const fileName = FILE_PREFIX + ts + '.json';
  folder.createFile(fileName, JSON.stringify(payload), MimeType.PLAIN_TEXT);

  const list = getSortedBackupFiles(folder);
  if (list.length > MAX_BACKUPS) {
    for (let i = MAX_BACKUPS; i < list.length; i++) {
      list[i].setTrashed(true);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, fileName: fileName })).setMimeType(ContentService.MimeType.JSON);
}
